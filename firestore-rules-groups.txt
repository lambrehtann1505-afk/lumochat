ПРАВИЛА FIRESTORE ДЛЯ ГРУПП И КАНАЛОВ
=====================================
Firebase Console -> Firestore Database -> Правила (Rules).
Внутри существующего блока  match /databases/{database}/documents { ... }
ДОБАВЬТЕ (не удаляя старые правила) эти два блока и нажмите «Опубликовать»:

    match /handles/{h} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow delete: if request.auth != null && resource.data.owner == request.auth.uid;
    }

    match /groups/{gid} {
      allow read: if request.auth != null;

      allow create: if request.auth != null
        && request.resource.data.owner == request.auth.uid
        && request.auth.uid in request.resource.data.members;

      // владелец и админы могут менять всё;
      // остальные: вступить/выйти сами или обновить «последнее сообщение»
      allow update: if request.auth != null && (
        resource.data.owner == request.auth.uid
        || request.auth.uid in resource.data.admins
        || request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['last', 'lastTs', 'lastFrom', 'lastName'])
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members'])
            && request.auth.uid in request.resource.data.members
            && request.resource.data.members.size() == resource.data.members.size() + 1
            && request.resource.data.members.hasAll(resource.data.members))
        || (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['members'])
            && !(request.auth.uid in request.resource.data.members)
            && request.resource.data.members.size() == resource.data.members.size() - 1
            && resource.data.members.hasAll(request.resource.data.members))
      );

      allow delete: if request.auth != null && resource.data.owner == request.auth.uid;

      match /messages/{mid} {
        allow read: if request.auth != null;
        allow create: if request.auth != null
          && request.resource.data.from == request.auth.uid
          && request.auth.uid in get(/databases/$(database)/documents/groups/$(gid)).data.members
          && (get(/databases/$(database)/documents/groups/$(gid)).data.type == 'group'
              || request.auth.uid in get(/databases/$(database)/documents/groups/$(gid)).data.admins);
        allow delete: if request.auth != null && (
          resource.data.from == request.auth.uid
          || request.auth.uid in get(/databases/$(database)/documents/groups/$(gid)).data.admins);
      }
    }
